cmake_minimum_required(VERSION 3.20)
project(ai_tool_policy LANGUAGES CXX)

add_library(ai_tool_policy src/policy.cpp)
target_include_directories(ai_tool_policy PUBLIC include)
target_compile_features(ai_tool_policy PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(ai_tool_policy PRIVATE /W4 /WX)
else()
    target_compile_options(ai_tool_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(policy_tests tests/policy_test.cpp)
    target_link_libraries(policy_tests PRIVATE ai_tool_policy)
    if(MSVC)
        target_compile_options(policy_tests PRIVATE /W4 /WX)
    else()
        target_compile_options(policy_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME policy_tests COMMAND policy_tests)
endif()
